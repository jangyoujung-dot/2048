// 기존의 move 함수들을 아래와 같이 수정하면 정확하게 작동합니다. function moveLeft() { comboCount = 0; let before = JSON.stringify(grid); grid = grid.map(r => slide(r)); if (before !== JSON.stringify(grid)) { randomCell(); showCombo(); update(); checkGameOver(); } } function moveRight() { let before = JSON.stringify(grid); grid = grid.map(r => slide([...r].reverse()).reverse()); // 좌우 반전 후 슬라이드 if (before !== JSON.stringify(grid)) { randomCell(); showCombo(); update(); checkGameOver(); } } function moveUp() { let before = JSON.stringify(grid); rotate(); // 90도 회전 rotate(); rotate(); // 결과적으로 270도 회전 (위쪽이 왼쪽으로 오게 함) grid = grid.map(r => slide(r)); rotate(); // 90도 회전하여 원상복구 if (before !== JSON.stringify(grid)) { randomCell(); showCombo(); update(); checkGameOver(); } } function moveDown() { let before = JSON.stringify(grid); rotate(); // 90도 회전 (아래쪽이 왼쪽으로 오게 함) grid = grid.map(r => slide(r)); rotate(); rotate(); rotate(); // 270도 회전하여 원상복구 if (before !== JSON.stringify(grid)) { randomCell(); showCombo(); update(); checkGameOver(); } }
